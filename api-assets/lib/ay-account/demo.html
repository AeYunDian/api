AyAccount SDK Demo

在同一个页面里演示全屏 iframe、嵌入式容器、主题、语言、关闭弹窗和宿主 Toast。

全屏 iframe 会覆盖本页按钮；打开后可用浮动条实时切换语言和主题。

嵌入式容器：点击“嵌入式登录”后在这里加载 SDK

查看调用代码 
<script src="https://online.undz.cn/lib/ay-account/main.js?v=3.2.0"></script>
<script>
"use strict";
const account = new AyAccount({
  appId: "ayaccountcenter_1601",
  theme: "auto" // auto | light | dark
});

// 全屏 iframe
account.login().then((result) => console.log(result));

// 嵌入式
account.login({ container: "#inlineSlot", minHeight: "600px" });

// 注册 / 主题 / 语言
account.register({ theme: "dark" });
account.setTheme("light");
account.changeLanguage("en-us");

// 首屏进入动画（默认 back；可分别配置移动端/电脑端）
account.login({ enter: "back" });
account.login({ enter: { mobile: "up", desktop: "next" } });
account.login({ enter: "none" });

// 嵌入式卡片默认撑满容器；也可手动关闭
account.login({ container: "#inlineSlot", fill: true });
account.login({ container: "#inlineSlot", fill: false });

// 更多窗口配置
account.login({ only: "login", hideFaq: true, hideSwitch: true });
account.register({ only: "register" });
account.login({ hideFaq: true });
account.register({ hideSwitch: true });
account.login({ scrim: false }); // 电脑端全屏去掉半透明遮罩
</script>
